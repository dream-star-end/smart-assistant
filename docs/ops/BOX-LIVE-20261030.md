## 已执行只读SQL（脱敏）

全部使用个人库openclaude_v5_selfhost，本地postgres socket、default_transaction_read_only=on、20秒timeout、BEGIN READ ONLY/ROLLBACK；无数据库URL、ID、用户内容、账号标识。原始聚合见管理工作区 `rf-boxlive/readonly-results.json`，原/新指标见metrics-before/after JSON与MD。

```sql
-- models
BEGIN READ ONLY; SET LOCAL TIME ZONE 'UTC'; SET LOCAL statement_timeout='20000';
SELECT coalesce(json_agg(t),'[]'::json) FROM (WITH known AS (SELECT DISTINCT model FROM turn_dispatches WHERE model LIKE 'box-%'), h AS (SELECT * FROM turn_dispatches WHERE admitted_at >= '2026-10-28T20:15:58Z'::timestamptz AND admitted_at < '2026-10-29T16:35:19Z'::timestamptz AND user_id<>765 AND session_id LIKE 'web%' AND client_message_id LIKE 'm-%' AND client_message_id NOT LIKE 'm-recover-%' AND model LIKE 'box-%') SELECT k.model, CASE WHEN k.model LIKE 'box-api-claude-%' THEN 'model_api' WHEN k.model LIKE 'box-claude-%' THEN 'interactive' ELSE 'other_box' END AS route_family, count(h.dispatch_id) AS turns,count(*) FILTER(WHERE h.outcome='completed') AS completed,count(*) FILTER(WHERE h.dispatch_id IS NOT NULL AND h.outcome IS DISTINCT FROM 'completed') AS non_completed,count(*) FILTER(WHERE h.outcome='not_accepted') AS not_accepted,count(*) FILTER(WHERE h.dispatch_id IS NOT NULL AND h.status IS DISTINCT FROM 'terminal') AS non_terminal FROM known k LEFT JOIN h ON h.model=k.model GROUP BY k.model ORDER BY k.model) t;
ROLLBACK;

-- failure_buckets
BEGIN READ ONLY; SET LOCAL TIME ZONE 'UTC'; SET LOCAL statement_timeout='20000';
SELECT coalesce(json_agg(t),'[]'::json) FROM (SELECT model,CASE WHEN model LIKE 'box-api-claude-%' THEN 'model_api' WHEN model LIKE 'box-claude-%' THEN 'interactive' ELSE 'other_box' END AS route_family,CASE WHEN outcome='not_accepted' THEN 'admission' WHEN status IS DISTINCT FROM 'terminal' THEN 'in_flight' ELSE 'execution' END AS derived_stage,status,outcome,coalesce(failure_code,'(none)') AS failure_code,count(*) AS n FROM turn_dispatches WHERE admitted_at >= '2026-10-28T20:15:58Z'::timestamptz AND admitted_at < '2026-10-29T16:35:19Z'::timestamptz AND user_id<>765 AND session_id LIKE 'web%' AND client_message_id LIKE 'm-%' AND client_message_id NOT LIKE 'm-recover-%' AND model LIKE 'box-%' AND outcome IS DISTINCT FROM 'completed' GROUP BY 1,2,3,4,5,6 ORDER BY n DESC,model) t;
ROLLBACK;

-- populations
BEGIN READ ONLY; SET LOCAL TIME ZONE 'UTC'; SET LOCAL statement_timeout='20000';
SELECT coalesce(json_agg(t),'[]'::json) FROM (SELECT CASE WHEN user_id=765 THEN 'acceptance' WHEN session_id NOT LIKE 'web%' THEN 'automation' WHEN client_message_id LIKE 'm-recover-%' THEN 'recover' WHEN client_message_id LIKE 'm-%' THEN 'human' ELSE 'other' END AS cohort,count(*) AS n FROM turn_dispatches WHERE admitted_at >= '2026-10-28T20:15:58Z'::timestamptz AND admitted_at < '2026-10-29T16:35:19Z'::timestamptz AND model LIKE 'box-%' GROUP BY 1 ORDER BY 1) t;
ROLLBACK;

-- journal_buckets
BEGIN READ ONLY; SET LOCAL TIME ZONE 'UTC'; SET LOCAL statement_timeout='20000';
SELECT coalesce(json_agg(t),'[]'::json) FROM (SELECT ctx->>'model' AS model,ctx->>'boxInvocationMode' AS invocation_mode,ctx->>'boxState' AS box_state,coalesce(ctx->>'boxUnknownPhase','(unrecorded)') AS phase,state,coalesce(failure_code,'(none)') AS failure_code,coalesce(ctx->'boxExpiredClose'->>'cause','(none)') AS close_cause,count(*) AS n FROM request_finalize_journal WHERE ctx ? 'boxState' AND created_at >= '2026-10-28T20:15:58Z'::timestamptz AND created_at < '2026-10-29T16:35:19Z'::timestamptz AND user_id<>765 GROUP BY 1,2,3,4,5,6,7 ORDER BY model) t;
ROLLBACK;

-- liveness
BEGIN READ ONLY; SET LOCAL TIME ZONE 'UTC'; SET LOCAL statement_timeout='20000';
SELECT coalesce(json_agg(t),'[]'::json) FROM (SELECT count(*) AS journals,max(created_at) AS last_created,count(*) FILTER(WHERE created_at >= '2026-10-28T20:15:58Z'::timestamptz AND created_at < '2026-10-29T16:35:19Z'::timestamptz) AS window_journals,count(*) FILTER(WHERE ctx ? 'boxRunAbsent') AS absence_records,count(*) FILTER(WHERE ctx->'boxExpiredClose'->>'cause'='box_process_absent') AS absent_closed,count(*) FILTER(WHERE ctx->>'boxState' IN ('handoff','unknown','resuming')) AS active_leaves FROM request_finalize_journal WHERE ctx ? 'boxState' AND created_at < '2026-10-29T16:35:19Z'::timestamptz) t;
ROLLBACK;

-- continuation_decisions
BEGIN READ ONLY; SET LOCAL TIME ZONE 'UTC'; SET LOCAL statement_timeout='20000';
SELECT coalesce(json_agg(t),'[]'::json) FROM (SELECT bucket_start,component,control_plane_release,control_plane_commit,split_part(labels->>'kind',':',1) AS stage,split_part(labels->>'kind',':',2) AS decision,split_part(labels->>'kind',':',3) AS reason,sum(sample_count) AS n FROM telemetry_metric_rollups WHERE metric='box_continuation_decisions' AND bucket_start < '2026-10-29T16:35:19Z'::timestamptz AND bucket_start + interval '5 minutes' > '2026-10-28T20:15:58Z'::timestamptz GROUP BY 1,2,3,4,5,6,7 ORDER BY 1,2,5,6,7) t;
ROLLBACK;

-- after_login_fix
BEGIN READ ONLY; SET LOCAL TIME ZONE 'UTC'; SET LOCAL statement_timeout='20000';
SELECT coalesce(json_agg(t),'[]'::json) FROM (SELECT model,count(*) AS turns,count(*) FILTER(WHERE outcome='completed') AS completed,count(*) FILTER(WHERE outcome IS DISTINCT FROM 'completed') AS non_completed FROM turn_dispatches WHERE admitted_at >= '2026-10-29T10:55:39Z'::timestamptz AND admitted_at < '2026-10-29T16:35:19Z'::timestamptz AND user_id<>765 AND session_id LIKE 'web%' AND client_message_id LIKE 'm-%' AND client_message_id NOT LIKE 'm-recover-%' AND model LIKE 'box-%' GROUP BY model) t;
ROLLBACK;
```
